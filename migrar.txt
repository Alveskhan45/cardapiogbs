PLANO DE MIGRAÇÃO: SINGLE-TENANT → MULTI-TENANT (SAAS)
Projeto: Cardápio Digital - Distribuidora de Bebidas
Data: 2026-10-04
================================================================================

VISÃO GERAL
-----------
Transformar a aplicação atual (um banco, um admin, uma instância) em uma
plataforma SaaS multi-tenant onde cada cliente (distribuidora) tem seus
dados isolados, configurações próprias e acesso via subdomínio.

ABORDAGEM ESCOLHIDA: Shared Database + Shared Schema (tenant_id em todas tabelas)
- Menor risco, migração incremental, um deploy só
- PostgreSQL recomendado para produção (SQLite só para dev)
- Isolamento via middleware + queries com WHERE tenant_id

2.1 Criar middleware: middleware/tenantResolver.js
    
    // Extrai tenant_id do request
    // Suporta: subdomain (cliente.seusaas.com), header (X-Tenant-ID), path (/t/cliente1/)
    
    function resolveTenant(req, res, next) {
      let tenantIdentifier = null;
      
      if (process.env.TENANT_RESOLVER === 'subdomain') {
        const host = req.get('host') || '';
        // Remove porta se houver
        const hostname = host.split(':')[0];
        // Extrai subdomínio: cliente.seusaas.com → cliente
        const parts = hostname.split('.');
        if (parts.length >= 3) {
          tenantIdentifier = parts[0];
        }
      } else if (process.env.TENANT_RESOLVER === 'header') {
        tenantIdentifier = req.get('X-Tenant-ID');
      } else if (process.env.TENANT_RESOLVER === 'path') {
        const match = req.path.match(/^\/t\/([^\/]+)/);
        if (match) tenantIdentifier = match[1];
      }
      
      if (!tenantIdentifier) {
        // Permitir acesso sem tenant para rotas públicas (landing, login super-admin)
        if (isPublicRoute(req.path)) return next();
        return res.status(400).json({ error: 'Tenant não identificado' });
      }
      
      // Buscar tenant no banco (com cache Redis em produção)
      db.get('SELECT id, name, subdomain, status, plan FROM tenants WHERE subdomain = ? AND status = "active"', 
        [tenantIdentifier], (err, tenant) => {
        if (err) return res.status(500).json({ error: 'Erro ao resolver tenant' });
        if (!tenant) return res.status(404).json({ error: 'Tenant não encontrado ou inativo' });
        
        req.tenant = tenant;
        req.tenantId = tenant.id;
        
        // Para RLS: definir variável de sessão
        if (process.env.USE_RLS === 'true') {
          db.run('SET LOCAL app.current_tenant_id = ?', [tenant.id]);
        }
        
        next();
      });
    }

3.3 Middleware de autorização:
    
    function requireAuth(req, res, next) {
      const auth = req.headers.authorization;
      if (!auth?.startsWith('Bearer ')) return res.status(401).json({ error: 'Token requerido' });
      
      try {
        const decoded = jwt.verify(auth.slice(7), process.env.JWT_SECRET);
        req.user = decoded;
        // Verificar se token pertence ao tenant da request
        if (!decoded.superAdmin && decoded.tenantId !== req.tenantId) {
          return res.status(403).json({ error: 'Acesso negado a este tenant' });
        }
        next();
      } catch (e) {
        return res.status(401).json({ error: 'Token inválido ou expirado' });
      }
    }
    
    function requireRole(...roles) {
      return (req, res, next) => {
        if (!req.user) return res.status(401).json({ error: 'Não autenticado' });
        if (req.user.superAdmin) return next();
        if (!roles.includes(req.user.role)) return res.status(403).json({ error: 'Permissão insuficiente' });
        next();
      };
    }

    async function provisionTenant(data) {
      const tenantId = uuid();
      const tx = db.transaction();
      
      try {
        // 1. Criar tenant
        tx.run('INSERT INTO tenants (id, name, subdomain, plan) VALUES (?, ?, ?, ?)',
          [tenantId, data.name, data.subdomain, data.plan || 'starter']);
        
        // 2. Criar admin user
        const passHash = await hashPassword(data.adminPassword || generatePassword());
        tx.run('INSERT INTO users (tenant_id, email, password_hash, name, role) VALUES (?, ?, ?, ?, "admin")',
          [tenantId, data.adminEmail, passHash, data.adminName]);
        
        // 3. Config padrão
        tx.run(`INSERT INTO config (tenant_id, store_name, delivery_time, min_order, 
                  payments, frete_mode, theme_primary) 
                VALUES (?, ?, ?, ?, ?, ?, ?)`,
          [tenantId, data.name, '30-45 min', 0, JSON.stringify(['Pix','Dinheiro','Cartão na entrega']),
           'bairro', '#d90429']);
        
        // 4. Dados demo (opcional): categorias, produtos, bairros exemplo
        await seedDemoData(tx, tenantId);
        
        // 5. Criar pasta assets
        fs.mkdirSync(path.join(DATA_DIR, 'tenants', tenantId, 'produtos'), { recursive: true });
        
        tx.commit();
        return { tenantId, adminPassword: data.adminPassword || generatedPass };
      } catch (e) {
        tx.rollback();
        throw e;
      }
    }

9.2 Middleware de verificação de limites:
    
    async function checkLimit(tenantId, resource, action = 'create') {
      const tenant = await db.get('SELECT plan FROM tenants WHERE id = ?', [tenantId]);
      const plan = await db.get('SELECT limits FROM plans WHERE id = ?', [tenant.plan]);
      const limit = plan.limits[resource];
      
      if (limit === -1) return { allowed: true }; // ilimitado
      
      const current = await db.get(`SELECT COUNT(*) as c FROM ${resource} WHERE tenant_id = ?`, [tenantId]);
      if (current.c >= limit) {
        return { allowed: false, limit, current: current.c, upgradeUrl: '/pricing' };
      }
      return { allowed: true, limit, current: current.c, remaining: limit - current.c };
    }
    
    // Usar em: POST /api/products, POST /api/bairros, POST /api/users, etc.
